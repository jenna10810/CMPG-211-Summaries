\section{SU2: Complexity}

\subsection{Introduction}

\begin{itemize}
    \item compare different algorithm complexity in order to develop more efficient programs
    \item two important aspects when comparing algorithms:
        \begin{enumerate}
            \item the \textcolor{Periwinkle}{memory} of memory used by the algorithm 
                \begin{itemize}
                    \item[] memory is determined by the chosen variables and can generally be determined by looking at the code
                \end{itemize}
            \item the \textcolor{Periwinkle}{running time} of the algorithm
                \begin{itemize}
                    \item[] running time depends on the algorithm and the number of data items processed by the algorithm 
                \end{itemize}
        \end{enumerate}
    \item the focus is on the \textbf{running time} of an algorithm - we are using algorithm analysis rather than machine runtime
\end{itemize}

        \begin{enumerate}

            \item the \textcolor{Periwinkle}{simplified model} where each tau is assigned the value equal to 1

\subsection{Detailed Model}

\begin{itemize}
    \item \textbf{basic assumption:} the time it takes to \emph{fetch} or \emph{store} a variable in memory is always the same
\end{itemize}

\begin{center}
    \begin{minipage}{0.20\textwidth}
        \begin{tightcode}
x = y; // or 
x = 1
        \end{tightcode}
    \end{minipage}
\end{center}


\begin{itemize}
    \item[]{}
        \begin{itemize}
            \item the analysis includes the time to \textbf{fetch}
            \item \mintinline{java}{y} is the first instance; or
            \item the constant \mintinline{java}{1} in the second instance
            \item \textbf{plus} the time to \textbf{store} this value in the variable/value \mintinline{java}{x}
            \item we \textbf{do not fetch} the variable \mintinline{java}{x} as well
            \item the analysis of this line: $\displaystyle \tau_{\text{fetch}} + \tau_{\text{store}}$
        \end{itemize}
\end{itemize}

\begin{center}
    \begin{tabular}{|
>{\columncolor{Periwinkle!10}}l| 
>{\columncolor{Periwinkle!10}}l|}

        $\tau_{\text{fetch}}$ & time to fetch the operand from memory\\
        $\tau_{\text{store}}$ & store the result in memory \\
        $\tau_{(+,-,\div,\times)}$ & time for basic arithmetic operations\\
        $\tau_{\leq}$ & time for all comparisons - we use the same symbol for $<$, $>$, ==, $\leq$, $\geq$\\
        $\tau_{\text{call}}$ & time to pass a parameter is constant and equal to $\tau_{\text{store}}$\\
        $\tau_{\text{return}}$ & time to call a method and the time it takes to return from a method\\
        $\tau_{[\cdot]}$ & time for the address calculation based on the index of an array\\
        \hline

\subsection{Asymptotic Analysis - \text{$O(n)$}}

\begin{itemize}
    \item the aim of this section is to show how detailed, and especially simplified analysis, is used to determine one single term for an algorithm
    \item expressed as $O(X)$, where $X$ is expressed in terms of $n$ - which is the number of items manipulated by the algorithm to be analysed
    \item from detailed $\rightarrow$ simplified $\rightarrow$ $O(n)$ is a progression in understanding - it is very hard to understand big $O$ without the other two 
    \item the result of simplified analysis is a line we show on a graph (we can compare various lines on the same graph) where each line represents the running time of a specific algorithm
    \item the cost of the algorithm is mainly determined by the \textcolor{Periwinkle}{power or $n$} in the equation $\Rightarrow n^2$ is worse than $n$
    \item to go from simplified to asymptotic, you just keep the largest power of $n$
\end{itemize}

\subsubsection{Big \text{$O$}}

\begin{mybox}[Periwinkle]{definition}
    \begin{itemize}
        \item[] consider a function $f(n)$ that is non-negative for all integers $n \geq 0$
        \item[] we say that "$f(n)$ is big $O$ $g(n)$" which we write $f(n) = O(g(n))$, 
        \item[] if there exists an integer $n_0$ and a constant $c > 0$ so that for all integers,
    \end{itemize}

    \begin{center}
         $n \leq n_0$, $f(n) \leq c g(n)$
    \end{center}
\end{mybox}

\begin{mybox}[Periwinkle]{properties}
    \begin{itemize}
        \item[] if $f_1(n) = O(g_1(n))$ and $f_2(n) = O(g_2(n))$
            \begin{itemize}
                \item [1.] $f_1(n) + f_2(n) = O(\text{max}(g_1(n)))$
                \item [2.] $f_1(n) \times f_2(n) = O(g_1(n) \times g_2(n))$
                \item [3.] If $f(n) = a_mn^m + a_{m-1}n^{m-1} + a_{m-2}n^{m-2} + \cdots + a_3n^3 + a_2n^2 + a_1n + a_0$ then,
            \end{itemize}
    \end{itemize}

    \vspace{-0.5cm}

    \begin{center}
        $f(n) = O(g(n^m)) $
    \end{center}
\end{mybox}

            result = result * x + a[i];
        }
        return result;
    }
}
        \end{tightcode}

\subsection{Rules for \text{\mintinline{java}{Java}}}

\begin{mybox}[Periwinkle]{iteration}
    \noindent
    \begin{minipage}{0.8\textwidth}
        the worst-case running time of a sequence of \mintinline{java}{Java} such as:\\
        is $O(\text{max}(T_1(n), T_2(n), \cdots, T_m(n)))$, where $S_i$ , the i$^{\text{th}}$ statement in the sequence, is $O(T_i(n))$
    \end{minipage}
    \begin{minipage}{0.03\textwidth}
        \phantom{text}
    \end{minipage}
    \begin{minipage}{0.12\textwidth}
        \begin{tightcode}
S_1;
S_2;
S_m;
        \end{tightcode}
    \end{minipage}
\end{mybox}


\begin{mybox}[Periwinkle]{sequential composition}
    \noindent
    \begin{minipage}{0.3\textwidth}
        \begin{center}
            \begin{tightcode}
for(S_1; S_2; S_3) {
    S_4;
}
            \end{tightcode}
        \end{center}
    \end{minipage}
    \begin{minipage}{0.05\textwidth}
        \phantom{text}
    \end{minipage}
    \begin{minipage}{0.6\textwidth}
        the worst-case running time of a \mintinline{java}{Java for} loop such as:\\
        is $O(\text{max}(T_1(n), T_2(n) \times (I(n) + 1), T_3 \times I(n), T_4(n) \times I(n)))$\\
        where the running time of statement $S_1$ is $O(T_i(n))$, for $i = 1,2,3,4$ and $I(n)$ is the number of iterations executed in the worst case
    \end{minipage}
\end{mybox}

\begin{mybox}[Periwinkle]{conditional execution}
    \noindent
    \begin{minipage}{0.7\textwidth}
        the worst-case running time of a \mintinline{java}{Java if-then-else} statement such as:\\
        is $O(\text{max}(T_1(1), T_2(n), T_3(n)))$, for $i = 1,2,3$
    \end{minipage}

        \phantom{text}
    \end{minipage}

if(S_1)
    S_2;

    S_3;
